# Схема БД

```sql
-- Здания (Лоджия 4, 5, 6, 7, 8)
CREATE TABLE buildings (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL UNIQUE,
    active INTEGER NOT NULL DEFAULT 1
);

-- Помещения здания
CREATE TABLE rooms (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    building_id INTEGER NOT NULL REFERENCES buildings(id) ON DELETE CASCADE,
    key TEXT NOT NULL,              -- restaurant, kitchen, toilet...
    name TEXT NOT NULL,             -- Ресторан, Кухня...
    required INTEGER NOT NULL DEFAULT 1,  -- сколько человек нужно
    gender TEXT NOT NULL DEFAULT '',       -- M / F / '' (любой)
    active INTEGER NOT NULL DEFAULT 1,
    UNIQUE(building_id, key)
);

-- Люди (привязаны к зданию)
CREATE TABLE people (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL UNIQUE,
    gender TEXT NOT NULL CHECK(gender IN ('M','F')),
    building_id INTEGER REFERENCES buildings(id),
    active INTEGER NOT NULL DEFAULT 1
);

-- Периоды дежурств (гибкая длительность)
CREATE TABLE periods (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,
    start_date TEXT NOT NULL,       -- YYYY-MM-DD
    num_days INTEGER NOT NULL DEFAULT 7,
    is_current INTEGER NOT NULL DEFAULT 0
);

-- Назначения на дежурство
CREATE TABLE assignments (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    period_id INTEGER NOT NULL REFERENCES periods(id) ON DELETE CASCADE,
    person_id INTEGER NOT NULL REFERENCES people(id) ON DELETE CASCADE,
    room_id INTEGER NOT NULL REFERENCES rooms(id) ON DELETE CASCADE,
    day INTEGER NOT NULL CHECK(day >= 0),  -- offset от start_date
    UNIQUE(period_id, room_id, day, person_id)
);

-- Отсутствия на период
CREATE TABLE absences (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    person_id INTEGER NOT NULL REFERENCES people(id) ON DELETE CASCADE,
    period_id INTEGER NOT NULL REFERENCES periods(id) ON DELETE CASCADE,
    reason TEXT NOT NULL DEFAULT 'absent',  -- absent / sick / other
    UNIQUE(person_id, period_id)
);
```
